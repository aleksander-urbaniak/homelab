@pve-node01

global_defs {
    router_id pve-node01
    enable_script_security
    script_user root
}

vrrp_script chk_pve {
    script "/root/scripts/check-pve.sh"
    interval 5
    timeout 4
    fall 2
    rise 2
    weight -50
}

vrrp_instance VI_PVE {
    state BACKUP
    interface vmbr0
    virtual_router_id 51
    priority 100
    advert_int 1

    unicast_src_ip 192.0.2.11
    unicast_peer {
        192.0.2.12
        192.0.2.13
    }

    authentication {
        auth_type PASS
        auth_pass CHANGE_ME_KEEPALIVED_PASS
    }

    virtual_ipaddress {
        192.0.2.10/24 dev vmbr0
    }

    track_script {
        chk_pve
    }
}
